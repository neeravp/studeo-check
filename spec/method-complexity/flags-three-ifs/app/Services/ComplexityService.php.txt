<?php

namespace App\Services;

final class ComplexityService
{
    public function calculate(int $a, int $b, int $c): int
    {
        if ($a > 0) {
            return 1;
        } elseif ($b > 0) {
            return 2;
        } elseif ($c > 0) {
            return 3;
        }

        return 0;
    }
}
