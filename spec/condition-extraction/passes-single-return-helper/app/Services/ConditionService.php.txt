<?php

namespace App\Services;

final class ConditionService
{
    public function verify(bool $a, bool $b): int
    {
        if ($this->isValid($a, $b)) {
            return 1;
        }

        return 0;
    }

    private function isValid(bool $a, bool $b): bool
    {
        return $a && $b;
    }
}
